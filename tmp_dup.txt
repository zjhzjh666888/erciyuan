  getSpriteScreenPosition(charId: string): { x: number; y: number } | null {
    if (!this.game || !this.townScene) return null
    const world = this.townScene.getSpriteWorldPosition(charId)
    if (!world) return null

    // Phaser 涓荤浉鏈猴細world 鈫?camera-view锛堜粛鏄?Phaser game 鍐呴儴 px锛夈€?
    const camera = this.game.scene.scenes
      .find((s) => s.cameras?.main)
      ?.cameras.main
    if (!camera) return null
    const viewX = (world.x - camera.scrollX) * camera.zoom
    const viewY = (world.y - camera.scrollY) * camera.zoom

    // game 鍐呴儴 px 鈫?DOM CSS px锛欶IT 妯″紡涓嬬殑 displaySize / gameSize 姣斻€?
    const scale = this.game.scale
    const gameW = scale.gameSize.width
    const gameH = scale.gameSize.height
    const displayW = scale.displaySize.width
    const displayH = scale.displaySize.height
    if (gameW <= 0 || gameH <= 0 || displayW <= 0 || displayH <= 0) return null
    const scaleX = displayW / gameW
    const scaleY = displayH / gameH

    // canvas 鍦?parent 瀹瑰櫒鍐呯殑灞呬腑鍋忕Щ锛團IT 鐣?letterbox锛夈€?
    const parent = scale.parent as HTMLElement | null
    if (!parent) return null
    const parentW = parent.clientWidth
    const parentH = parent.clientHeight
    const offsetX = (parentW - displayW) / 2
    const offsetY = (parentH - displayH) / 2

    return {
      x: offsetX + viewX * scaleX,
      y: offsetY + viewY * scaleY,
    }
  }
